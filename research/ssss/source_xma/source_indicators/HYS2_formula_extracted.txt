{参数设置：SCQH (0, 2, 0)  <-- 0:A股, 1:美股, 2:加密货币}
N:=15; M1:=3; M2:=3; PLONG:=33; PSHORT:=13;
O00:=(CLOSE-LLV(LOW,N))/(HHV(HIGH,N)-LLV(LOW,N))*100;

L1:SMA(O00,M1,1),COLORFF8D1E,LINETHICK1;
L2:SMA(L1,M2,1),COLOR0CAEE6,LINETHICK1;
L3:3*L1-2*L2,COLORE970DC,LINETHICK1;
B1:20,COLORGRAY,DOTLINE;
B2:80,COLORGRAY,DOTLINE;

O01:=REF((LOW+OPEN+CLOSE+HIGH)/4,1);
O02:=SMA(ABS(LOW-O01),PSHORT,1)/SMA(MAX(LOW-O01,0),10,1);
O03:=EMA(O02,10);
O04:=LLV(LOW,PLONG);
O05:=EMA(IF(LOW<=O04,O03,0),3);
O06:=SMA(ABS(HIGH-O01),PSHORT,1)/SMA(MIN(HIGH-O01,0),10,1);
O07:=EMA(O06,10);
O08:=HHV(HIGH,PLONG);
O09:=EMA(IF(HIGH>=O08,O07,0),3);

{=================== 第一步：提取公共微观恐慌因子 ===================}
VARB:=REF(LOW,1);
VARC:=SMA(ABS(LOW-VARB),3,1)/SMA(MAX(LOW-VARB,0),3,1)*100;
VAR7E:=LLV(LOW,30);
VAR7H:=IF(MA(CLOSE,58),1,0);

{=================== 第二步：分市场独立计算 ===================}

{--- 模式 0：A股日线深度沉淀模式（原版高沉淀参数） ---}
A_V1:=EMA(HHV(HIGH,500),21);
A_V2:=EMA(HHV(HIGH,250),21);
A_V3:=EMA(HHV(HIGH,90),21);
A_V4:=EMA(LLV(LOW,500),21);
A_V5:=EMA(LLV(LOW,250),21);
A_V6:=EMA(LLV(LOW,90),21);
A_V7:=EMA((A_V4*0.96+A_V5*0.96+A_V6*0.96+A_V1*0.558+A_V2*0.558+A_V3*0.558)/6,21);
A_V8:=EMA((A_V4*1.25+A_V5*1.23+A_V6*1.2+A_V1*0.55+A_V2*0.55+A_V3*0.65)/6,21);
A_V9:=EMA((A_V4*1.3+A_V5*1.3+A_V6*1.3+A_V1*0.68+A_V2*0.68+A_V3*0.68)/6,21);
A_VA:=EMA((A_V7*3+A_V8*2+A_V9)/6*1.738,21);
A_VD:=EMA(IF(CLOSE*1.35<=A_VA,VARC*10,VARC/10),3);
A_V7F:=HHV(A_VD,30);
A_底:=EMA(IF(LOW<=VAR7E,(A_VD+A_V7F*2)/2,0),3)/618*VAR7H;
A_VG:=HHV(HIGH,30);
A_顶:=EMA(IF(HIGH>=A_VG,-(A_VD+A_V7F*2)/2,0),3)/618*VAR7H;

{--- 模式 1：美股日线防钝化模式（周期减半，恐慌收敛） ---}
B_V1:=EMA(HHV(HIGH,250),21);
B_V2:=EMA(HHV(HIGH,120),21);
B_V3:=EMA(HHV(HIGH,60),21);
B_V4:=EMA(LLV(LOW,250),21);
B_V5:=EMA(LLV(LOW,120),21);
B_V6:=EMA(LLV(LOW,60),21);
B_V7:=EMA((B_V4*0.96+B_V5*0.96+B_V6*0.96+B_V1*0.558+B_V2*0.558+B_V3*0.558)/6,21);
B_V8:=EMA((B_V4*1.25+B_V5*1.23+B_V6*1.2+B_V1*0.55+B_V2*0.55+B_V3*0.65)/6,21);
B_V9:=EMA((B_V4*1.3+B_V5*1.3+B_V6*1.3+B_V1*0.68+B_V2*0.68+B_V3*0.68)/6,21);
B_VA:=EMA((B_V7*3+B_V8*2+B_V9)/6*1.738,21);
B_VD:=EMA(IF(CLOSE*1.30<=B_VA,VARC*5,VARC/10),3);
B_V7F:=HHV(B_VD,30);
B_底:=EMA(IF(LOW<=VAR7E,(B_VD+B_V7F*2)/2,0),3)/618*VAR7H;
B_VG:=HHV(HIGH,30);
B_顶:=EMA(IF(HIGH>=B_VG,-(B_VD+B_V7F*2)/2,0),3)/618*VAR7H;

{--- 模式 2：加密市场高频插针模式（极速短周期，恐慌拉满） ---}
C_V1:=EMA(HHV(HIGH,120),21);
C_V2:=EMA(HHV(HIGH,60),21);
C_V3:=EMA(HHV(HIGH,30),21);
C_V4:=EMA(LLV(LOW,120),21);
C_V5:=EMA(LLV(LOW,60),21);
C_V6:=EMA(LLV(LOW,30),21);
C_V7:=EMA((C_V4*0.96+C_V5*0.96+C_V6*0.96+C_V1*0.558+C_V2*0.558+C_V3*0.558)/6,21);
C_V8:=EMA((C_V4*1.25+C_V5*1.23+C_V6*1.2+C_V1*0.55+C_V2*0.55+C_V3*0.65)/6,21);
C_V9:=EMA((C_V4*1.3+C_V5*1.3+C_V6*1.3+C_V1*0.68+C_V2*0.68+C_V3*0.68)/6,21);
C_VA:=EMA((C_V7*3+C_V8*2+C_V9)/6*1.738,21);
C_VD:=EMA(IF(CLOSE*1.35<=C_VA,VARC*18,VARC/10),3);
C_V7F:=HHV(C_VD,30);
C_底:=EMA(IF(LOW<=VAR7E,(C_VD+C_V7F*2)/2,0),3)/618*VAR7H;
C_VG:=HHV(HIGH,30);
C_顶:=EMA(IF(HIGH>=C_VG,-(C_VD+C_V7F*2)/2,0),3)/618*VAR7H;

火焰山底:IF(SCQH=0,A_底,IF(SCQH=1,B_底,C_底)),COLORBLUE,NODRAW;
火焰山顶:IF(SCQH=0,A_顶,IF(SCQH=1,B_顶,C_顶)),COLORCCCC00,NODRAW;

今量:=火焰山底;
STICKLINE(火焰山底>0,0 ,今量*1.2,2,0 ),COLOR00AACC;
STICKLINE(火焰山底>0,0 ,今量*0.8,2,0 ),COLOR0088CC;
STICKLINE(火焰山底>0,0 ,今量*0.6,2,0 ),COLOR0066CC;
STICKLINE(火焰山底>0,0 ,今量*0.4,2,0 ),COLOR0033CC;
STICKLINE(火焰山底>0,0 ,今量*0.2,2,0 ),COLOR0000CC;
STICKLINE(火焰山顶<0,0,火焰山顶,2,0),COLORCCCC00;

O0A:=EMA(CLOSE,12) - EMA(CLOSE,26);
O0B:=EMA(O0A,9);
O0C:=2*(O0A-O0B);
O0D:=O0C > REF(O0C,1);
O0E:=L2 < 55;
O0F:=L2 < 45;
O0G:=L2 > 70;

DRAWICON(CROSS(L1,L2) AND O0E, L1*0.9, 1);
DRAWICON(CROSS(L2,L1) AND O0G, L2*1.1, 2);
DRAWICON(CROSS(L1,L2) AND O0D AND O0F, L1*0.8, 25);
DRAWTEXT(CROSS(L1,L2) AND O0D AND O0F, L1*0.7, '★共振'), COLORBLUE;
